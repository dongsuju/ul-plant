#!/bin/bash

# 定义一些常用函数
info() {
  echo "[$(date +"%Y-%m-%d %H:%M:%S")] [INFO] $*"
}

warn() {
  echo "[$(date +"%Y-%m-%d %H:%M:%S")] [WARN] $*"
}

error() {
  echo "[$(date +"%Y-%m-%d %H:%M:%S")] [ERROR] $*"
}

fatal() {
  echo "[$(date +"%Y-%m-%d %H:%M:%S")] [FATAL] $*"
  exit 1
}

step=1
logStep() {
  info "====Step ${step} $*===="
  step=$((step + 1))
}

log_exit_code() {
  exit_code=$?
  if test $exit_code -ne 0; then
    warn "exec fail, exit code: ${exit_code}"
    return
  fi
  info "exec succ, exit code: ${exit_code}"
}

cd "$(dirname "/")" || exit
cur_dir=$(pwd)

# 0. set env
setBash() {
  info "set bash"
  bashrc=/root/.bashrc
  if ! test -f "${bashrc}.bak"; then
    info "backup bash"
    cp "${bashrc}" "${bashrc}.bak"
  fi

  banner='# history cfg add by instance tools'

  # 100 line => 4.4K, 100000 line ~> 4.4M

  if ! grep "${banner}" "${bashrc}"; then
    info "append bash"
    echo "${banner}"'
HISTCONTROL="ignorespace"
HISTTIMEFORMAT="%Y-%m-%d %H:%M:%S "
HISTFILESIZE=100000
HISTSIZE=1000

shopt -s histappend
PROMPT_COMMAND="history -a;$PROMPT_COMMAND"' >>"${bashrc}"
  fi
}

exportEnv() {
  OMP_NUM_THREADS=$(cat /proc/cpuinfo | grep processor | wc -l)
  export OMP_NUM_THREADS
}

removeOnethingService() {
  info "remove legacy onething services"
  onethingService=$(find /etc/supervisor/conf.d -name "onething-service-*.conf")
  for i in ${onethingService}; do
    info "remove service $i"
    rm -f "${i}"
    log_exit_code
  done
}

# support disable pkg
isPkgDisable() {
  # return 0 if disable
  pkgName="$1"
  if test -z "${pkgName}"; then
    return 1
  fi
  disableFlag="/etc/.disable_pkg_${pkgName}"
  if test -f "${disableFlag}"; then
    return 0
  fi
  return 1
}

pkgInit() {
  info "call pkg-init.sh"
  pkgPath="/.onething/instance/tools/pkg/"
  for pkg in "${pkgPath}"*; do
    info "find pkg ${pkg}"
    if isPkgDisable "${pkg#"${pkgPath}"}"; then
      info "disable pkg ${pkg}"
    else
      pkgInit="${pkg}/pkg-init.sh"
      info "call ${pkgInit}"
      timeout 5m sh "${pkgInit}"
      log_exit_code
    fi
  done
}

isServiceDisable() {
  # return 0 if disable
  service="$1"
  if test -z "${service}"; then
    return 1
  fi

  disableFlag="/etc/.disable_plugin_${service}"
  if test -f "${disableFlag}"; then
    return 0
  fi
